import React, { useState, useRef, useEffect } from "react";

interface PageData {
  id: number;
  content: string;
  color: string;
}

const PAGES: PageData[] = [
  { id: 1, content: "실제 종이처럼", color: "#ff6b6b" },
  { id: 2, content: "스프링을 따라", color: "#4dadf7" },
  { id: 3, content: "우측부터 입체적으로", color: "#51cf66" },
  { id: 4, content: "말려 넘어가는 효과", color: "#cc5de8" },
];

const WIDTH = 320;
const HEIGHT = 480;
const DURATION = 1700; // ms (전체 비행 기준 속도)
// 진행도가 이 지점에 닿으면 종이가 완전히 투명해짐 → 오브젝트 제거 후 액션 종료
const FADE_DONE = 0.4;

// 던지기 모션: 빠르게 출발한 뒤 공기저항처럼 감속 (easeOut)
function easeOutCubic(t: number) {
  return 1 - Math.pow(1 - t, 3);
}

// 스프링에서 뜯겨 나간 윗변의 너덜너덜한 톱니(찢김) 클립
const TORN_TOP_CLIP = (() => {
  const teeth = 26;
  const pts: string[] = ["0% 100%"];
  for (let i = 0; i <= teeth; i++) {
    const x = (i / teeth) * 100;
    const y = i % 2 === 0 ? 4 : 0.5;
    pts.push(`${x.toFixed(2)}% ${y}%`);
  }
  pts.push("100% 100%");
  return `polygon(${pts.join(", ")})`;
})();

export default function VerticalFlipBookV2() {
  const [index, setIndex] = useState(0);
  const [tearing, setTearing] = useState<{ from: number; to: number } | null>(
    null,
  );
  const [progress, setProgress] = useState(0);
  const rafRef = useRef<number | null>(null);
  const lockRef = useRef(false); // 동기적 애니메이션 잠금 (빠른 연타 방지)

  const startTear = () => {
    if (lockRef.current) return;
    const to = index + 1;
    if (to >= PAGES.length) return;
    lockRef.current = true;

    if (rafRef.current != null) cancelAnimationFrame(rafRef.current);

    setTearing({ from: index, to });
    setProgress(0);

    const start = performance.now();
    const step = (now: number) => {
      const raw = Math.min(1, (now - start) / DURATION);
      setProgress(raw);
      if (raw < FADE_DONE) {
        rafRef.current = requestAnimationFrame(step);
      } else {
        // 완전히 투명해짐 → 날아간 종이 제거 + 액션 종료
        rafRef.current = null;
        setIndex(to);
        setTearing(null);
        setProgress(0);
        lockRef.current = false;
      }
    };
    rafRef.current = requestAnimationFrame(step);
  };

  const goPrev = () => {
    if (lockRef.current) return;
    setIndex((i) => Math.max(0, i - 1));
  };

  useEffect(() => {
    return () => {
      if (rafRef.current != null) cancelAnimationFrame(rafRef.current);
      lockRef.current = false;
    };
  }, []);

  // 📝 종이 질감(줄노트 선)을 포함한 페이지 렌더러 (기존 목업과 동일)
  const renderPage = (data: PageData, showContent = true) => (
    <div
      style={{
        width: "100%",
        height: "100%",
        display: "flex",
        alignItems: "center",
        justifyContent: "center",
        backgroundColor: data.color,
        // 미세한 가로줄 패턴으로 종이 느낌 부여
        backgroundImage:
          "linear-gradient(rgba(255,255,255,0.08) 1px, transparent 1px)",
        backgroundSize: "100% 24px",
        color: "white",
        fontSize: 22,
        fontWeight: 700,
        textAlign: "center",
        padding: "40px 24px 0 24px", // 스프링 공간 확보를 위한 상단 패딩
        boxSizing: "border-box",
        textShadow: "0 2px 4px rgba(0,0,0,0.15)",
      }}
    >
      {showContent ? data.content : null}
    </div>
  );

  // 전환 중에는 뜯기는 종이 밑에서 다음 페이지가 즉시 드러남
  const baseIndex = tearing ? tearing.to : index;
  const base = PAGES[baseIndex];
  const tearingPage = tearing ? PAGES[tearing.from] : null;

  // 1시 방향(수직 기준 약 30°)으로 날아감: 위로 많이, 우측으로 약간.
  const eased = easeOutCubic(progress); // 비행 거리(공기저항처럼 감속)
  const flyX = eased * 460; // 우측 이동
  const flyY = eased * -760; // 상단(노트 밖)으로 이동
  const flyRot = eased * 38; // 던져지며 시계방향 회전
  const flyScale = 1 - eased * 0.22; // 멀어지며 살짝 축소
  // 멀어질수록 선형으로 점점 옅어져 FADE_DONE 시점에 완전 투명
  const flyOpacity = Math.max(0, 1 - progress / FADE_DONE);

  return (
    <div
      style={{
        display: "flex",
        flexDirection: "column",
        alignItems: "center",
        gap: 28,
        marginTop: 50,
      }}
    >
      {/* 메인 랩퍼 컨테이너 (기존 노트 모양 그대로) */}
      <div
        style={{
          width: WIDTH,
          height: HEIGHT,
          position: "relative",
          borderRadius: 16,
          boxShadow: "0 20px 50px rgba(0,0,0,0.3)",
          background: "#151515",
          perspective: 1500,
        }}
      >
        {/* 1️⃣ 노트를 고정하는 상단 입체 스프링 철사 UI */}
        <div
          style={{
            position: "absolute",
            top: 12,
            left: 20,
            right: 20,
            height: 16,
            display: "flex",
            justifyContent: "space-between",
            zIndex: 10,
            pointerEvents: "none",
          }}
        >
          {Array.from({ length: 9 }).map((_, i) => (
            <div
              key={i}
              style={{
                width: 8,
                height: 24,
                borderRadius: 4,
                background:
                  "linear-gradient(90deg, #555 0%, #ccc 50%, #444 100%)",
                boxShadow: "0 4px 6px rgba(0,0,0,0.4)",
              }}
            />
          ))}
        </div>

        {/* 날아가는 종이가 노트 밖으로 벗어날 수 있도록 클리핑은 바닥 페이지에만 적용 */}
        <div
          style={{
            position: "absolute",
            inset: 0,
            borderRadius: 16,
            overflow: "hidden",
            zIndex: 1,
          }}
        >
          {/* 바닥(현재/다음) 페이지 */}
          <div style={{ position: "absolute", inset: 0 }}>
            {renderPage(base)}
          </div>
        </div>

        {/* 2️⃣ 뜯겨 나가 1시 방향으로 날아가는 종이 (스프링·노트 위로 떠서 이동) */}
        {tearing && tearingPage && (
          <div
            style={{
              position: "absolute",
              inset: 0,
              zIndex: 20,
              clipPath: TORN_TOP_CLIP,
              transformOrigin: "50% 0%",
              transform: `translate(${flyX}px, ${flyY}px) rotate(${flyRot}deg) scale(${flyScale})`,
              opacity: flyOpacity,
              filter: "drop-shadow(-10px 14px 18px rgba(0,0,0,0.35))",
              willChange: "transform, opacity",
            }}
          >
            {renderPage(tearingPage)}
          </div>
        )}
      </div>

      {/* 하단 컨트롤 바 */}
      <div style={{ display: "flex", gap: 12, alignItems: "center" }}>
        <button onClick={goPrev} disabled={index === 0 || !!tearing} style={btnStyle}>
          이전
        </button>
        <span
          style={{
            color: "#aaa",
            fontSize: 14,
            minWidth: 48,
            textAlign: "center",
            fontWeight: 600,
          }}
        >
          {index + 1} / {PAGES.length}
        </span>
        <button
          onClick={startTear}
          disabled={index === PAGES.length - 1 || !!tearing}
          style={btnStyle}
        >
          뜯기
        </button>
      </div>
    </div>
  );
}

const btnStyle: React.CSSProperties = {
  padding: "12px 24px",
  fontSize: 16,
  fontWeight: "bold",
  cursor: "pointer",
  backgroundColor: "#252525",
  color: "white",
  border: "none",
  borderRadius: 8,
  outline: "none",
  boxShadow: "0 4px 10px rgba(0,0,0,0.2)",
  transition: "background-color 0.2s",
};
