import React, { useState, useRef, useEffect } from "react";

interface PageData {
  id: number;
  content: string;
  color: string;
}

const PAGES: PageData[] = [
  { id: 1, content: "실제 종이처럼", color: "#ff6b6b" },
  { id: 2, content: "스프링을 따라", color: "#4dadf7" },
  { id: 3, content: "우측부터 입체적으로", color: "#51cf66" },
  { id: 4, content: "말려 넘어가는 효과", color: "#cc5de8" },
];

const WIDTH = 320;
const HEIGHT = 480;
const DURATION = 950; // ms
// 좌우 컬 진행 곡선의 지수. 둘 다 1 미만(concave)이라 시작과 동시에 즉시 출발 → 초반 느림 없음.
// 오른쪽 지수가 더 작아 더 빠르게 선행하고, 둘 다 t=1에서 1로 수렴 → 상단 스프링에 동시 도달.
const RIGHT_EXP = 0.5;
const LEFT_EXP = 0.7;

function easeInOutCubic(t: number) {
  return t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2;
}

function clamp01(v: number) {
  return Math.max(0, Math.min(1, v));
}

function taper(t: number) {
  return Math.min(1, t * 5, (1 - t) * 5);
}

type Direction = "next" | "prev";

export default function EnhancedVerticalCurlFlipBook() {
  const [index, setIndex] = useState(0);
  const [transition, setTransition] = useState<{
    from: number;
    to: number;
    dir: Direction;
  } | null>(null);
  const [rawProgress, setRawProgress] = useState(0);
  const rafRef = useRef<number | null>(null);

  const startTransition = (dir: Direction) => {
    if (transition) return;
    const to = dir === "next" ? index + 1 : index - 1;
    if (to < 0 || to >= PAGES.length) return;

    setTransition({ from: index, to, dir });
    setRawProgress(0);

    const startTime = performance.now();
    const step = (now: number) => {
      const raw = Math.min(1, (now - startTime) / DURATION);
      setRawProgress(easeInOutCubic(raw));
      if (raw < 1) {
        rafRef.current = requestAnimationFrame(step);
      } else {
        setIndex(to);
        setTransition(null);
        setRawProgress(0);
      }
    };
    rafRef.current = requestAnimationFrame(step);
  };

  useEffect(() => {
    return () => {
      if (rafRef.current) cancelAnimationFrame(rafRef.current);
    };
  }, []);

  // 📝 종이 질감(줄노트 선)을 포함한 페이지 렌더러
  const renderPage = (data: PageData, showContent = true) => (
    <div
      style={{
        width: "100%",
        height: "100%",
        display: "flex",
        alignItems: "center",
        justifyContent: "center",
        backgroundColor: data.color,
        // 미세한 가로줄 패턴으로 종이 느낌 부여
        backgroundImage:
          "linear-gradient(rgba(255,255,255,0.08) 1px, transparent 1px)",
        backgroundSize: "100% 24px",
        color: "white",
        fontSize: 22,
        fontWeight: 700,
        textAlign: "center",
        padding: "40px 24px 0 24px", // 스프링 공간 확보를 위한 상단 패딩
        boxSizing: "border-box",
        textShadow: "0 2px 4px rgba(0,0,0,0.15)",
      }}
    >
      {showContent ? data.content : null}
    </div>
  );

  const dir = transition?.dir;
  const baseIndex = transition
    ? dir === "next"
      ? transition.to
      : transition.from
    : index;
  const animIndex = transition
    ? dir === "next"
      ? transition.from
      : transition.to
    : null;
  const base = PAGES[baseIndex];
  const animPage = animIndex !== null ? PAGES[animIndex] : null;

  const rolledAmount = !transition
    ? 0
    : dir === "next"
      ? rawProgress
      : 1 - rawProgress;

  // 좌우 모두 지수<1(concave)이라 시작 즉시 출발 → 왼쪽 초반 느림 제거.
  // 오른쪽 지수가 더 작아 선행하고, 둘 다 t=1에서 1에 도달 → 상단 스프링 동시 도착.
  const rolledRight = clamp01(Math.pow(rolledAmount, RIGHT_EXP));
  const rolledLeft = clamp01(Math.pow(rolledAmount, LEFT_EXP));

  const flatFracRight = 1 - rolledRight;
  const flatFracLeft = 1 - rolledLeft;
  const flatHeightRight = HEIGHT * flatFracRight;
  const flatHeightLeft = HEIGHT * flatFracLeft;

  const rollTopAvg = (flatHeightRight + flatHeightLeft) / 2;
  // 클립 컷(오른쪽 변이 더 위로 올라감)과 롤 바/그림자의 기울기를 동일 방향으로 맞춤.
  // CSS skewY 양수는 오른쪽을 아래로 내리므로, 오른쪽이 더 말린 상태를 표현하려면 부호를 반대로.
  const skewDeg =
    (Math.atan2(flatHeightRight - flatHeightLeft, WIDTH) * 180) / Math.PI;

  return (
    <div
      style={{
        display: "flex",
        flexDirection: "column",
        alignItems: "center",
        gap: 28,
        marginTop: 50,
      }}
    >
      {/* 메인 랩퍼 컨테이너 */}
      <div
        style={{
          width: WIDTH,
          height: HEIGHT,
          position: "relative",
          borderRadius: 16,
          boxShadow: "0 20px 50px rgba(0,0,0,0.3)",
          background: "#151515",
          perspective: 1500,
        }}
      >
        {/* 1️⃣ 노트를 고정하는 상단 입체 스프링 철사 UI */}
        <div
          style={{
            position: "absolute",
            top: 12,
            left: 20,
            right: 20,
            height: 16,
            display: "flex",
            justifyContent: "space-between",
            zIndex: 10,
            pointerEvents: "none",
          }}
        >
          {Array.from({ length: 9 }).map((_, i) => (
            <div
              key={i}
              style={{
                width: 8,
                height: 24,
                borderRadius: 4,
                background:
                  "linear-gradient(90deg, #555 0%, #ccc 50%, #444 100%)",
                boxShadow: "0 4px 6px rgba(0,0,0,0.4)",
              }}
            />
          ))}
        </div>

        {/* 내부 가림막 보정 (둥근 모서리 클리핑용) */}
        <div
          style={{
            position: "absolute",
            inset: 0,
            borderRadius: 16,
            overflow: "hidden",
          }}
        >
          {/* 바닥에 대기 중인 페이지 */}
          <div style={{ position: "absolute", inset: 0, zIndex: 1 }}>
            {renderPage(base)}
          </div>

          {transition && animPage && (
            <>
              {/* 평평하게 깎여나가는 남은 종이 파트 */}
              <div
                style={{
                  position: "absolute",
                  top: 0,
                  left: 0,
                  width: "100%",
                  height: HEIGHT,
                  zIndex: 3,
                  clipPath: `polygon(0% 0%, 100% 0%, 100% ${flatFracRight * 100}%, 0% ${flatFracLeft * 100}%)`,
                }}
              >
                {renderPage(animPage)}
              </div>

              {/* 분할선 아래 종이(남은 종이의 하단부)를 분할선 기준으로 수직 대칭시켜 위로 접어올린 플랩 */}
              <div
                style={{
                  position: "absolute",
                  top: 0,
                  left: 0,
                  width: "100%",
                  height: HEIGHT,
                  zIndex: 4,
                  // 분할선 아래쪽 밴드만 남김 (넘어가는 종이의 하단부)
                  clipPath: `polygon(0% ${flatFracLeft * 100}%, 100% ${flatFracRight * 100}%, 100% 100%, 0% 100%)`,
                  // 사선 분할선(중점 50%·rollTopAvg, 각도 skewDeg)을 축으로 반사.
                  // 임의 각도 α 직선 반사 = rotate(α)·scaleY(-1)·rotate(-α). 수평 반사로 하면
                  // 사선 기울기 부호가 뒤집혀 컷 라인과 반대로 기울며 X자가 생기므로 반드시 사선축 반사.
                  transformOrigin: `50% ${rollTopAvg}px`,
                  transform: `rotate(${skewDeg}deg) scaleY(-1) rotate(${-skewDeg}deg)`,
                  // 접힌 종이가 평면부 위로 드리우는 그림자
                  filter: "drop-shadow(0 0 14px rgba(0,0,0,0.4))",
                  backfaceVisibility: "visible",
                }}
              >
                {/* 종이 뒷면이므로 글자는 숨기고 색/질감만 표시 */}
                {renderPage(animPage, false)}

                {/* 접혀 올라간 뒷면 음영: 크레스트(접힌 모서리)는 살짝 밝고 자유단(끝)으로 갈수록 짙어짐 */}
                <div
                  style={{
                    position: "absolute",
                    inset: 0,
                    pointerEvents: "none",
                    mixBlendMode: "multiply",
                    background: `linear-gradient(to bottom,
                      rgba(255,255,255,0.35) 0%,
                      rgba(0,0,0,0.15) 8%,
                      rgba(0,0,0,0.32) 55%,
                      rgba(0,0,0,0.5) 100%)`,
                  }}
                />
              </div>

              {/* 분할선(접힘 골)에서 바로 시작하는 연속 음영: 위(플랩)·아래 음영을 이어 빨간 빈틈 제거 */}
              <div
                style={{
                  position: "absolute",
                  left: 0,
                  width: "100%",
                  top: rollTopAvg,
                  height: 44,
                  zIndex: 5,
                  transform: `skewY(${skewDeg}deg)`,
                  transformOrigin: "50% 0%",
                  // 분할선에서 가장 어둡고 아래로 갈수록 옅어짐. taper로 완전히 말리면 소멸.
                  background: `linear-gradient(180deg, rgba(0,0,0,${0.5 * taper(rolledAmount)}) 0%, rgba(0,0,0,0) 100%)`,
                  pointerEvents: "none",
                }}
              />
            </>
          )}
        </div>
      </div>

      {/* 하단 컨트롤 바 */}
      <div style={{ display: "flex", gap: 12, alignItems: "center" }}>
        <button
          onClick={() => startTransition("prev")}
          disabled={index === 0 || !!transition}
          style={btnStyle}
        >
          이전
        </button>
        <span
          style={{
            color: "#aaa",
            fontSize: 14,
            minWidth: 48,
            textAlign: "center",
            fontWeight: 600,
          }}
        >
          {index + 1} / {PAGES.length}
        </span>
        <button
          onClick={() => startTransition("next")}
          disabled={index === PAGES.length - 1 || !!transition}
          style={btnStyle}
        >
          다음
        </button>
      </div>
    </div>
  );
}

const btnStyle: React.CSSProperties = {
  padding: "12px 24px",
  fontSize: 16,
  fontWeight: "bold",
  cursor: "pointer",
  backgroundColor: "#252525",
  color: "white",
  border: "none",
  borderRadius: 8,
  outline: "none",
  boxShadow: "0 4px 10px rgba(0,0,0,0.2)",
  transition: "background-color 0.2s",
};
